*** 服务构造函数
对 shepherd 服务的三种高频形态进行封装，来方便配置文件全局进行调用。

=command= 为「程序 + 参数」的 argv 列表：第一个元素是程序，其余每个元素是一个参数，逐段列出而非写成一整条命令字符串。元素可以是字符串，也可以是可 lower 对象——构建期统一展开为具体值后拼成真正的命令行：

- 字符串： ="/bin/sh"= 、 ="--bind-dynamic"= -> 原样使用；
- 可 lower 对象：
   =(file-append kdeconnect "/bin/kdeconnect-indicator")= -> 展开为 =/gnu/store/…= 下的绝对路径
   =(local-file "usb-power-watch")= -> 指向本地文件
   gexp =#~(…)= -> 在构建期求值。
因此无需手写 =/gnu/store= 路径，直接引用包对象即可。

+ =home-daemon-service= ：用户常驻进程，日志统一写入到 =$XDG_STATE_HOME/shepherd/<name>.log= 中。会自动继承调用时的用户环境。
#+NAME: home-daemon-service-helpers
#+begin_src scheme
(define* (home-daemon-service name command
                              #:key (requirement '())
                              (auto-start? #t) (respawn? #t)
                              (documentation "[No documentation.]"))
  (shepherd-service
   (provision (list name))
   (requirement requirement)
   (documentation documentation)
   (start #~(lambda args
              ((make-forkexec-constructor
                (list #$@command)
                #:environment-variables (environ)
                #:log-file (string-append (getenv "XDG_STATE_HOME")
                                          "/shepherd/" #$(symbol->string name)
                                          ".log"))
               args)))
   (stop #~(make-kill-destructor))
   (auto-start? auto-start?)
   (respawn? respawn?)))
#+end_src

+ =root-daemon-service= ：系统常驻进程，支持 =#:log-file= 与 =#:one-shot?= 。
#+NAME: root-daemon-service-helpers
#+begin_src scheme
(define* (root-daemon-service name command
                              #:key (requirement '())
                              (auto-start? #t) (respawn? #t)
                              one-shot? log-file
                              (documentation "[No documentation.]"))
  (simple-service name shepherd-root-service-type
                  (list (shepherd-service
                         (provision (list name))
                         (requirement requirement)
                         (documentation documentation)
                         (one-shot? one-shot?)
                         (auto-start? auto-start?)
                         (respawn? respawn?)
                         (start #~(make-forkexec-constructor
                                   (list #$@command)
                                   #$@(if log-file (list #:log-file log-file) '())))
                         (stop #~(make-kill-destructor))))))
#+end_src

+ =root-one-shot-service= ：开机跑一次即退出（one-shot + 不 respawn）， =start= / =#:stop= 直接传 gexp。
  respawn?/stop 默认值与 shepherd-service 记录一致（one-shot 服务两者均惰性，不写死 #f 以保持与手写定义零漂移）。

                         (requirement requirement)
                         (documentation documentation)

** FileSystems
:PROPERTIES:
:CUSTOM_ID: sys-filesystems
:END:

存储采用 =tmpfs on rootfs= 作为基础，利用 =LUKS 加密的 Btrfs Subvolume= 来持久化需要的数据：
- 根目录 =/= 挂载在 tmpfs，关机即清空。
- 关键系统数据与用户目录持久化存储于 Btrfs 子卷中，直接挂载到对应目录，子卷清单与挂载点定义于 [[./information.scm][information.scm]] 。
- 休眠到独立 swap 分区（由于 Guile initrd 有限制，所以只能采用没有加密的 swap 恢复）。

#+NAME: filesystem-defines
#+begin_src scheme
(define %swap-uuid "169557cc-00a4-448c-9bb6-7bd80fc2b023")

(define %mapped-devices-config
  (list (mapped-device
         (source (uuid "327f2e02-1e4f-48b2-87f0-797c481850c9"))
         (target "root")
         (type luks-device-mapping))))

(define %tmpfs
  (list (file-system
         (mount-point "/")
         (device "tmpfs")
         (type "tmpfs")
         (flags '(no-atime))
         (options "mode=0755,nr_inodes=1m,size=10%")
         (needed-for-boot? #t)
         (check? #f))
        (file-system
         (device "tmpfs")
         (mount-point "/var/lock")
         (type "tmpfs")
         (flags '(no-suid no-dev strict-atime))
         (options "mode=1777,nr_inodes=100k,size=1%")
         (needed-for-boot? #t)
         (create-mount-point? #t)
         (check? #f))))

(define %persist-filesystem
  (map (match-lambda
         ((subvol mount-point)
          (file-system
           (device (file-system-label "Linux"))
           (mount-point mount-point)
           (type "btrfs")
           (dependencies %mapped-devices-config)
           (options (string-append "subvol=" subvol ",compress=zstd:3"))
           (create-mount-point? #t)
           (needed-for-boot? (not (string=? mount-point "/tmp")))
           (check? (string=? mount-point "/gnu"))))) %btrfs-subvolumes))

(define %data-filesystem
  (file-system
   (device (file-system-label "Linux"))
   (mount-point "/data")
   (type "btrfs")
   (dependencies %mapped-devices-config)
   (options (string-append "subvol=" %btrfs-subvol-data ",compress=zstd:3"))
   (create-mount-point? #t)))

(define %bind-mounts
  (map (lambda (dirs)
         (file-system
          (mount-point (string-append "/home/" username "/" dirs))
          (device (string-append "/data/" dirs))
          (type "none")
          (flags '(bind-mount))
          (options "x-gvfs-hide")
          (dependencies (list %data-filesystem))
          (create-mount-point? #t))) %data-dirs))

(define %file-systems-config
  (append %bind-mounts %persist-filesystem %tmpfs
          (list %data-filesystem
                (file-system
                 (device (uuid "9699-52A2"
                               'fat))
                 (mount-point "/efi")
                 (type "vfat")
                 (create-mount-point? #t))) %base-file-systems))

(define %swap-devices
  (list (swap-space
         (target (uuid %swap-uuid)))))

**** NM dispatcher 脚本
WiFi 连上可信 profile 时把 =wlp0s20f3= 加入 nftables 中的 =trusted_ifaces= 集合，断开或不可信时移除；
tangle 到 =tmp/= 后由 [[#main-skeleton][服务装配]] 注入 =@VAR@= 占位符，利用 activation 拷贝到 =/etc/NetworkManager/dispatcher.d/= 。

#+NAME: nm-dispatcher-trusted-connection
#+begin_src sh :tangle ../tmp/nm-dispatcher-trusted-connection
#!/bin/sh

AGE_BIN="@AGE_BIN@"
AGE_IDENTITY="@AGE_IDENTITY@"
GREP_BIN="@GREP_BIN@"
NFT_BIN="@NFT_BIN@"
TRUSTED_CONNECTIONS_CIPHER="@TRUSTED_CONNECTIONS_CIPHER@"
WIFI_IFACE="wlp0s20f3"

ACTION="$2"
CONNECTION_ID="${CONNECTION_ID:-}"

[ "$1" = "$WIFI_IFACE" ] || exit 0

is_trusted_connection() {
    [ -r "$AGE_IDENTITY" ] || return 1
    "$AGE_BIN" -d -i "$AGE_IDENTITY" "$TRUSTED_CONNECTIONS_CIPHER" 2>/dev/null |
        "$GREP_BIN" -Fqx -- "$1"
}

remove_wifi_from_trusted_set() {
    "$NFT_BIN" delete element inet filter trusted_ifaces { "$WIFI_IFACE" } 2>/dev/null || true
}

add_wifi_to_trusted_set() {
    remove_wifi_from_trusted_set
    "$NFT_BIN" add element inet filter trusted_ifaces { "$WIFI_IFACE" }
}

case "$ACTION" in
    up)
        if is_trusted_connection "$CONNECTION_ID"; then
            add_wifi_to_trusted_set
        else
            remove_wifi_from_trusted_set
        fi
        ;;
    down)
        remove_wifi_from_trusted_set
        ;;
esac
#+end_src

信道探测： =iw link= 输出只有 freq（MHz），按频段换算信道号——2.4G: ch=(freq-2407)/5，5G: ch=(freq-5000)/5；未连接回落 ch6。
信道号决定频段与 hw_mode：1-14 是 2.4G（g），其余 5G（a）。

#+begin_src sh :tangle ../tmp/hotspot-ap-run
sta_channel() {
    freq=$("$IW_BIN" dev "$WIFI_IFACE" link 2>/dev/null |
               "$SED_BIN" -n 's/^[[:space:]]*freq: \([0-9][0-9]*\)$/\1/p')
    case "$freq" in
        "") return 1 ;;
        *) if [ "$freq" -ge 5000 ] 2>/dev/null; then
               echo $(( (freq - 5000) / 5 ))
           else
               echo $(( (freq - 2407) / 5 ))
           fi ;;
    esac
}

ch=$(sta_channel) || ch=6

if [ "$ch" -le 14 ] 2>/dev/null; then
    hw_mode=g
else
    hw_mode=a
fi

creds=$("$AGE_BIN" -d -i "$AGE_IDENTITY" "$HOTSPOT_CREDS_CIPHER" 2>/dev/null) || exit 1
ssid=$(printf '%s\n' "$creds" | "$SED_BIN" -n '1p')
psk=$(printf '%s\n' "$creds" | "$SED_BIN" -n '2p')
[ -n "$ssid" ] && [ -n "$psk" ] || exit 1

cur_type=$("$IW_BIN" dev "$HOTSPOT_IFACE" info 2>/dev/null |
               "$SED_BIN" -n 's/^[[:space:]]*type \([a-z_]*\)$/\1/p')
if [ "$cur_type" != "__ap" ]; then
    "$IP_BIN" link set "$HOTSPOT_IFACE" down 2>/dev/null
    "$IW_BIN" dev "$HOTSPOT_IFACE" set type __ap || exit 1
fi

"$MKDIR_BIN" -p "$CONF_DIR"
umask 077
{
    echo "interface=$HOTSPOT_IFACE"
    echo "driver=nl80211"
    echo "country_code=CN"
    echo "ieee80211d=1"
    echo "ssid=$ssid"
    echo "hw_mode=$hw_mode"
    echo "channel=$ch"
    echo "wpa=2"
    echo "wpa_passphrase=$psk"
    echo "wpa_key_mgmt=WPA-PSK"
    echo "rsn_pairwise=CCMP"
    echo "auth_algs=1"
    echo "macaddr_acl=0"
} > "$conf"

"$IP_BIN" addr flush dev "$HOTSPOT_IFACE" 2>/dev/null
"$IP_BIN" addr add "$AP_ADDR" dev "$HOTSPOT_IFACE" || exit 1
"$IP_BIN" link set "$HOTSPOT_IFACE" up
exec "$HOSTAPD_BIN" "$conf"

**** AC 电源状态联动
插电时切换至高性能模式（PPD performance + 关闭 WiFi 省电 + 提升 CPU 最低主频 + 启用 ASPM 性能策略）；拔电时切换至节能策略。开机由 =ac-power-init= 初始化检测。

#+NAME: ac-power-services
#+begin_src scheme
(let* ([ppdctl (file-append (spec->pkg "power-profiles-daemon")
                            "/bin/powerprofilesctl")]
       [iw-bin (file-append (spec->pkg "iw") "/sbin/iw")]

       [ac-power-script
        (program-file
         "ac-power-profile"
         #~(let ((mode (if (> (length (command-line)) 1)
                           (cadr (command-line))
                           ""))
                 (ppdctl #$ppdctl)
                 (iw-bin #$iw-bin))
             (define (run . args) (apply system* args) #t)
             (define (set-sys path val)
               (false-if-exception
                (call-with-output-file path
                  (lambda (port) (display val port)))))
             (let ([ac? (string=? mode "on")])
               ;; userspace 守护
               (run ppdctl "set" (if ac? "performance" "power-saver"))
               (run iw-bin "dev" "wlp0s20f3" "set" "power_save"
                    (if ac? "off" "on"))
               ;; intel_pstate：AC 抬最低频消升频延迟 + IO 等待自动抬 EPP
               (set-sys "/sys/devices/system/cpu/intel_pstate/min_perf_pct"
                        (if ac? "30" "8"))
               (set-sys "/sys/devices/system/cpu/intel_pstate/hwp_dynamic_boost"
                        (if ac? "1" "0"))
               ;; workqueue 就近派发 / 定时器本地唤醒
               (set-sys "/sys/module/workqueue/parameters/power_efficient"
                        (if ac? "N" "Y"))
               (set-sys "/proc/sys/kernel/timer_migration"
                        (if ac? "0" "1"))
               ;; PCIe 链路电源策略
               (set-sys "/sys/module/pcie_aspm/parameters/policy"
                        (if ac? "performance" "powersupersave"))
               ;; PCI runtime PM 白名单：AC 常开不进 D3，电池恢复 auto
               (for-each
                (lambda (bdf)
                  (set-sys (string-append "/sys/bus/pci/devices/" bdf
                                          "/power/control")
                           (if ac? "on" "auto")))
                '("0000:00:02.0"   ; i915 iGPU
                  "0000:01:00.0"   ; YMTC PC300 NVMe
                  "0000:00:14.3")) ; iwlwifi wlp0s20f3
               ;; 声卡：AC 零延迟；电池恢复默认 10s 省电
               (set-sys "/sys/module/snd_hda_intel/parameters/power_save"
                        (if ac? "0" "10"))
               ;; KSM 同页合并：扫描 CPU 开销只在 AC 付
               (when ac?
                 (set-sys "/sys/kernel/mm/ksm/pages_to_scan" "200")
                 (set-sys "/sys/kernel/mm/ksm/sleep_millisecs" "50"))
               (set-sys "/sys/kernel/mm/ksm/run" (if ac? "1" "0"))
               ;; 周期中断源常关（nowatchdog cmdline 之外的运行时兜底）
               (set-sys "/proc/sys/kernel/nmi_watchdog" "0")
               (set-sys "/proc/sys/kernel/soft_watchdog" "0"))))]

       [ac-power-rules
        (file->udev-rule
         "90-ac-power-profile.rules"
         (mixed-text-file
          "90-ac-power-profile.rules"
          "ACTION==\"add|change\", SUBSYSTEM==\"power_supply\", "
          "ATTR{type}==\"Mains\", ATTR{online}==\"1\", "
          "RUN+=\"" ac-power-script " on\"\n"
          "ACTION==\"add|change\", SUBSYSTEM==\"power_supply\", "
          "ATTR{type}==\"Mains\", ATTR{online}==\"0\", "
          "RUN+=\"" ac-power-script " off\"\n"))])

  (list (root-one-shot-service 'ac-power-init '(power-profiles-daemon user-processes)
                               #~(lambda _
                                   (let* ((port (false-if-exception
                                                 (open-input-file
                                                  "/sys/class/power_supply/ADP1/online")))
                                          (val (and port (read port))))
                                     (when port (close-port port))
                                     (zero? (system* #$ac-power-script
                                                     (if (eq? val 1)
                                                         "on"
                                                         "off")))))
                               #:stop #~(lambda _ #t) #:respawn? #f
                               #:documentation
                               "按当前电源状态初始化 PPD profile 与 WiFi 省电。")

        (udev-rules-service 'ac-power-profile ac-power-rules)))

*** Kanata 键盘映射
笔记本内置键盘（AT Translated Set 2 keyboard）映射。
目标： *不插外接小键盘时，内置键盘也能给出一致的手感* 。小键盘的每一列对应标准键盘的一个行（列 4 ↔ QWERTY、列 3 ↔ ASDF、列 2 ↔ ZXCV、列 1 ↔ 底部行）
+ 这个小键盘的列首键依次是 Esc / Tab / Shift / Alt，因此修改内置键盘的 base 层：
  - QWERTY 行首（物理 Tab）→ Esc
  - ASDF 行首（物理 CapsLock）→ Tab（CapsLock 功能在 fn 层收回）
  - 底部行前两个（物理 Ctrl / Alt）互换

右 Alt 按住时激活 fn 层，复刻小键盘驱动定制后的 Fn 层：
  Q W E R T Y U I O P → 1 2 3 4 5 6 7 8 9 0      列 4 的 Fn 层是整条数字行
  A S D F G          → F1 F2 F3 F4 F5            列 3 的 Fn 层接管 F1-F5
  H J K L            → [ ] ; '                   列 2 的 Fn 层接管符号区域
  Z X C V B N M      → F6 F7 F8 F9 F10 F11 F12   列 2 的 Fn 层接管 F6-F12

原本的 Tab（即小键盘的 Esc 键位）→ `
Enter / Backspace  → \ / -

**** usb-power-watch 脚本
熄屏断电守护，挂在物理 USB 口上的设备无论是什么都一律挂起，清单外的端口不碰，清单内的端口熄屏写 auto、亮屏写 on 。

#+NAME: usb-power-watch
#+begin_src sh :tangle ../tmp/usb-power-watch
#!/bin/sh

INTERVAL=30
THRESHOLD=120

PORTS="3-2 3-5 3-6 4-5 4-6"
in_ports() {
    for p in $PORTS; do
        case "$1" in "$p"|"$p."*) return 0 ;; esac
    done
    return 1
}

set_ports() {
    for dev in /sys/bus/usb/devices/*-*; do
        case "${dev##*/}" in *:*) continue ;; esac   # 跳过接口目录
[ -f "$dev/power/control" ] || continue
if in_ports "${dev##*/}"; then
    echo "$1" > "$dev/power/control" 2>/dev/null
fi
done
}

 off_since=0
 state=on
 while :; do
     lit=0; dark=0
     for b in /sys/class/backlight/*/bl_power; do
         [ -r "$b" ] || continue
         if [ "$(cat "$b" 2>/dev/null)" = 0 ]; then lit=1; else dark=1; fi
     done
     if [ "$lit" = 0 ] && [ "$dark" = 1 ]; then
         now=$(date +%s)
         [ "$off_since" = 0 ] && off_since=$now
         if [ $((now - off_since)) -ge $THRESHOLD ]; then
             set_ports auto; state=off
         fi
     else
         off_since=0
         if [ "$state" != on ]; then set_ports on; state=on; fi
     fi
     sleep $INTERVAL
 done
#+end_src

:PROPERTIES:

自建 Live 安装镜像的 =operating-system= 声明，与主机 =%system= 独立。
