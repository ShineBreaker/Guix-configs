  (map (match-lambda
         ((subvol mount-point)
          (file-system
           (device (file-system-label "Linux"))
           (mount-point mount-point)
           (type "btrfs")
           (dependencies %mapped-devices-config)

  (map (lambda (dirs)

# Auto-generated by Guix config.org.

#      即自重入后台执行。
#   3. 多入口并发由 flock 串行化。
#   4. 应急开关：/run/hotspot-disabled 存在时只关不开（tmpfs，重启自愈）。
#
# 变量可用环境覆盖，便于离线真值表测试；udev 与 dispatcher 环境

# iw link 输出只有 freq（MHz），按频段换算信道号：
# 2.4G: ch=(freq-2407)/5，5G: ch=(freq-5000)/5
sta_channel() {
    freq=$("$IW_BIN" dev "$WIFI_IFACE" link 2>/dev/null |
           "$SED_BIN" -n 's/^[[:space:]]*freq: \([0-9][0-9]*\)$/\1/p')
    case "$freq" in
        "") return 1 ;;
        *) if [ "$freq" -ge 5000 ] 2>/dev/null; then
               echo $(( (freq - 5000) / 5 ))
           else
               echo $(( (freq - 2407) / 5 ))
           fi ;;
    esac
}

ap_channel() {
    "$SED_BIN" -n 's/^channel=\([0-9][0-9]*\)$/\1/p' "$HOSTAPD_CONF" 2>/dev/null
}

# 目标信道——热点的唯一合法信道来源（安全闸）：
#   1. STA 已连接：跟随 STA 信道；
#   2. STA 未连接：扫描缓存里可见的 autoconnect WiFi profile 的信道
#      （按信号序取首个命中），让 NM 稍后自动关联时天然同信道；
#   3. 都没有：返回失败，调用方必须放弃本次动作。
# nmcli -t 会把值里的冒号转义为 "\:"，前后两种输出的转义形式一致，
# 故全程用转义形式匹配。不做子串 grep（"Office" 会误吞 "HomeOffice:"），
# 改为对扫描行按信号序逐行做字面前缀剥离 + 纯数字校验。
target_channel() {
    ch=$(sta_channel) && { echo "$ch"; return 0; }
    scan=$("$NMCLI_BIN" -t -f SSID,CHAN device wifi list ifname "$WIFI_IFACE" 2>/dev/null) || return 1
    [ -n "$scan" ] || return 1
    autoconns=$("$NMCLI_BIN" -t -f NAME,TYPE,AUTOCONNECT connection show 2>/dev/null |
                "$SED_BIN" -n 's/:802-11-wireless:yes$//p')
    [ -n "$autoconns" ] || return 1
    while IFS= read -r sline; do
        [ -n "$sline" ] || continue
        while IFS= read -r name; do
            [ -n "$name" ] || continue
            ch=${sline#"$name":}
            case "$ch" in
                ''|*[!0-9]*) continue ;;
                *) ap_channel_ok "$ch" || continue
                   echo "$ch"; return 0 ;;
            esac
        done <<EONAMES
$autoconns
EONAMES
    done <<EOSCAN
$scan
EOSCAN
    return 1
}

# 信道允许 AP 主动辐射校验：本机 self-managed regdom（world）下 5G 全带
# no IR，扫描分支（STA 未连）起在这些信道会被 hostapd 拒绝（NO-IR）。
# 带 IR-CONCURRENT 的信道在 STA 关联后限制解除——跟随 STA 的分支不需要
# 此过滤，只有扫描分支（发起时 STA 未连）必须选无 no IR 标志的信道。
# iw phy 的 Frequencies 行格式：* 5785 MHz [157] (22.0 dBm) (no IR)
ap_channel_ok() {
    line=$("$IW_BIN" phy 2>/dev/null | "$GREP_BIN" -E "MHz \[$1\] ")
    [ -n "$line" ] || return 1
    case "$line" in
        *"no IR"*) return 1 ;;
        *) return 0 ;;
    esac
}

stop_hotspot() {
    "$HERD_BIN" stop "$HOTSPOT_SVC" 2>/dev/null
    "$IP_BIN" addr flush dev "$HOTSPOT_IFACE" 2>/dev/null
    return 0
}

if [ -e "$DISABLE_SENTINEL" ]; then
    hotspot_active && stop_hotspot
    exit 0
fi

if on_ac_power && ! wifi_trusted; then
    ch=$(target_channel) || exit 0
    if hotspot_active; then
        ap_ch=$(ap_channel)
        [ -n "$ap_ch" ] && [ "$ap_ch" = "$ch" ] && exit 0

    exec "$HERD_BIN" start "$HOTSPOT_SVC"
fi

# 的教训之一）、按信道段生成 hw_mode（hostapd 默认 g 只有信道 1-14，
# 5G 信道号配 g = "Could not determine operating frequency"，事故中
# 热点 8 连败的真凶）、从 age 密文解出 SSID/密码、纠正接口类型
# （managed 残留 → __ap，切换需先 down）、配置 uap0 的热点子网地址，
# 最后 exec hostapd 成为 shepherd 跟踪的主进程。凭据只经管道与
# /run（tmpfs）过境，不落持久盘。变量可用环境覆盖，便于离线测试。
HOTSPOT_IFACE="${HOTSPOT_IFACE:-uap0}"
AP_ADDR="${AP_ADDR:-192.168.77.1/24}"
CONF_DIR="${CONF_DIR:-/run/auto-hotspot}"
CHANNEL_FILE="${CHANNEL_FILE:-/run/auto-hotspot/channel}"
AGE_IDENTITY="${AGE_IDENTITY:-@AGE_IDENTITY@}"
HOTSPOT_CREDS_CIPHER="${HOTSPOT_CREDS_CIPHER:-@HOTSPOT_CREDS_CIPHER@}"
AGE_BIN="${AGE_BIN:-@AGE_BIN@}"
IW_BIN="${IW_BIN:-@IW_BIN@}"
IP_BIN="${IP_BIN:-@IP_BIN@}"
SED_BIN="${SED_BIN:-@SED_BIN@}"
MKDIR_BIN="${MKDIR_BIN:-@MKDIR_BIN@}"
HOSTAPD_BIN="${HOSTAPD_BIN:-@HOSTAPD_BIN@}"

conf="$CONF_DIR/hostapd.conf"

# 信道必须由决策脚本显式提供（安全闸在决策侧，此处是防御性校验）
ch=$(cat "$CHANNEL_FILE" 2>/dev/null)
case "$ch" in
    ''|*[!0-9]*) exit 1 ;;
esac
# 信道号决定频段与 hw_mode：1-14 是 2.4G（g），其余 5G（a）
if [ "$ch" -le 14 ] 2>/dev/null; then
    hw_mode=g
else
    hw_mode=a
fi

creds=$("$AGE_BIN" -d -i "$AGE_IDENTITY" "$HOTSPOT_CREDS_CIPHER" 2>/dev/null) || exit 1
ssid=$(printf '%s\n' "$creds" | "$SED_BIN" -n '1p')
psk=$(printf '%s\n' "$creds" | "$SED_BIN" -n '2p')
[ -n "$ssid" ] && [ -n "$psk" ] || exit 1

# 接口类型纠正：早期部署残留的 managed 类型会让 hostapd 起不来
cur_type=$("$IW_BIN" dev "$HOTSPOT_IFACE" info 2>/dev/null |
           "$SED_BIN" -n 's/^[[:space:]]*type \([a-z_]*\)$/\1/p')
if [ "$cur_type" != "__ap" ]; then
    "$IP_BIN" link set "$HOTSPOT_IFACE" down 2>/dev/null
    "$IW_BIN" dev "$HOTSPOT_IFACE" set type __ap || exit 1
fi

"$MKDIR_BIN" -p "$CONF_DIR"
umask 077
{
    echo "interface=$HOTSPOT_IFACE"
    echo "driver=nl80211"
    echo "country_code=CN"
    echo "ieee80211d=1"
    echo "ssid=$ssid"
    echo "hw_mode=$hw_mode"
    echo "channel=$ch"
    echo "wpa=2"
    echo "wpa_passphrase=$psk"
    echo "wpa_key_mgmt=WPA-PSK"
    echo "rsn_pairwise=CCMP"
    echo "auth_algs=1"
    echo "macaddr_acl=0"
} > "$conf"

"$IP_BIN" addr flush dev "$HOTSPOT_IFACE" 2>/dev/null
"$IP_BIN" addr add "$AP_ADDR" dev "$HOTSPOT_IFACE" || exit 1
"$IP_BIN" link set "$HOTSPOT_IFACE" up
exec "$HOSTAPD_BIN" "$conf"
#+end_src

usb-power-watch

#+name: usb-power-watch
#+begin_src sh :tangle ../tmp/usb-power-watch
#!/bin/sh
# Auto-generated by Guix config.org.

# 熄屏断电守护：轮询背光 bl_power（普通用户可读，灭屏 0→4），
# 全灭持续 THRESHOLD 秒 → 挂起清单端口上的 USB 设备；复亮立即恢复。
# 覆盖手动 power-off-monitors、合盖关屏等一切熄屏路径；外接无
# 背光屏不产生信号，取向安全侧（不断电）。只读 sysfs，不依赖
# 图形会话——避开 graphical-session 启动竞态导致的拉起失败。
THRESHOLD=120
INTERVAL=30

# 断电端口清单：挂在这些物理 USB 口上的设备（含 hub 子树）无论
# 是什么一律挂起，清单外端口不碰。名称为 /sys/bus/usb/devices/
# 下的设备名（bus-port）；同一物理口的 USB2/USB3 视图各占一个名
# 字（3-5 与 4-5 为同一口），空口列上无副作用。3-2 是外接 dock 的
# USB2 视图，键盘/鼠标等 USB2 外设挂其子树；USB3 视图 2-2 不列入
# ——网卡在 2-2.4，避免熄屏断网（USB3 设备不占 USB2 视图，互不影
# 响）。挂起的 HID 收到输入由内核自动恢复，事件不丢；不要把承
# 载远程链路的设备插进清单口。依赖 udev 规则
# 95-usb-power-control-perms.rules 已把 power/control 放开为 0666，
# 本脚本以普通用户身份运行。
PORTS="3-2 3-5 3-6 4-5 4-6"
in_ports() {
    for p in $PORTS; do
        case "$1" in "$p"|"$p."*) return 0 ;; esac
    done
    return 1
}

 # 清单端口熄屏写 auto、亮屏写 on，不看设备当前值也不记挂了谁：
 # 写入幂等，熄屏期每轮都执行，新插设备下一轮即被覆盖。
 set_ports() {
     for dev in /sys/bus/usb/devices/*-*; do
         case "${dev##*/}" in *:*) continue ;; esac   # 跳过接口目录
 [ -f "$dev/power/control" ] || continue
 if in_ports "${dev##*/}"; then
     echo "$1" > "$dev/power/control" 2>/dev/null
 fi
done
 }

  off_since=0
  state=on
  while :; do
      lit=0; dark=0
      for b in /sys/class/backlight/*/bl_power; do
          [ -r "$b" ] || continue
          if [ "$(cat "$b" 2>/dev/null)" = 0 ]; then lit=1; else dark=1; fi
      done
      if [ "$lit" = 0 ] && [ "$dark" = 1 ]; then
          now=$(date +%s)
          [ "$off_since" = 0 ] && off_since=$now
          if [ $((now - off_since)) -ge $THRESHOLD ]; then
              set_ports auto; state=off
          fi
      else
          off_since=0
          if [ "$state" != on ]; then set_ports on; state=on; fi
      fi
      sleep $INTERVAL
  done
#+end_src
